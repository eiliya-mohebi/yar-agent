# backend/yar/app.py — respond() core
with self.tracer.turn(user_message):
    system = self.session.build_system(user_message, notify=notify)
    messages = (
        [{"role": "system", "content": system}]
        + self.session.history[-window:]
        + [{"role": "user", "content": user_message}]
    )
    result = run_loop(..., messages=messages, tools=self.tools, observer=notify)
    self.session.add_exchange(user_message, result.reply, tool_calls=..., source=source, meta=meta)
    self.memory.maybe_consolidate(notify=notify)
    self.memory.export_markdown()
return result
